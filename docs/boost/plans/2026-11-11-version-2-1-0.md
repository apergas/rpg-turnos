# Versión 2.1.0: plan de implementación

> **For agentic workers:** REQUIRED SUB-SKILL: Use boost:subagent-driven-development (recommended) or boost:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** La segunda zona de la v2: el desierto. Un mapa abierto de 90 × 70 m con dunas, cactus, un oasis con palmeras y sus dos pozos, y una ruina donde espera el Gusano de las dunas; cinco enemigos nuevos, el equipo de oricalco en la tienda de la Caravana (tramo 2), dos misiones (`escorpiones` y `cantimplora`, con un viajero nuevo), su fondo de combate, su música, su ambiente y sus partículas. La regla de los grupos de monstruos pasa a "fuera del camino" en las zonas sin hierba. Spec: `docs/boost/specs/2026-11-10-version-2-0-0-design.md` (M-43 a M-46, el desierto; "Versiones", fila 2.1.0; "Partida automática"; "Verificación"). Lo que ese spec dejó abierto lo fija `docs/boost/specs/2026-11-11-version-2-1-0-design.md`.

**Architecture:** Casi todo es dato sobre las reglas de la 2.0.0 y la 2.0.1. Lo nuevo en código: `ZoneData` gana `dunes`, `tree_areas`, `ground_patches`, `path_color` y `hill_color`, y `WorldStage` los construye (dunas sólidas con su aviso, palmeras solo dentro de las áreas); `MonsterGroups` deja de pedir hierba; `Decor` / `NatureModels` ganan los cactus, los matojos y las piezas de la ruina (con un tinte por material de arenisca); `World` gana el tipo de figura `item` ("Coger"); `BattleActor` gana cuatro figuras y tres variantes; `BattleStage` el fondo `desierto`; `StoryState` generaliza el "aún no" de la v2; `Game` gana `--autoplay-zone=desierto` y entrega las misiones que esperan en una tienda.

**Tech Stack:** Godot 4.7.2 (Compatibility), GDScript, escenas como texto, web PWA de un solo hilo.

## Global Constraints

- **Sin tests de ningún tipo, nunca.** Nada de TDD, ni carpeta `tests/`, ni frameworks (GUT, gdUnit4), ni pasos de "escribe un test que falle". Se verifica a mano: `--check-only` en cada script cambiado, `--import` después de añadir un `class_name` o modelos, partidas automáticas (`-- --autoplay`, `--autoplay-zone=...`), la salida de Godot, y fotos o fotogramas (`--write-movie`, o un script del scratchpad que guarda `root.get_texture().get_image()`). Los scripts de comprobación viven en el scratchpad, nunca en el repositorio.
- **Godot solo a través de `C:\Users\peixe\Documents\rpg-turnos\tools\godot.cmd`.** **Toda** ejecución (también `--import`, `--check-only` y los scripts del scratchpad) lleva `APPDATA` redirigido **en el mismo comando** a una carpeta literal del scratchpad, desde Git Bash: `APPDATA='<scratchpad>\impl\appdata' ./tools/godot.cmd ...`. `<scratchpad>` es siempre la ruta literal y completa del scratchpad de la sesión que implementa (por ejemplo `C:\Users\peixe\AppData\Local\Temp\rpg-scratch`): se escribe entera en cada orden. La partida real del usuario vive en `%APPDATA%\Godot\app_userdata\rpg-turnos\` y no se toca nunca: nunca se usa `$APPDATA` en `rm`, `mv`, `cp` ni en una ruta, y nunca se borra nada fuera del scratchpad. La única excepción es exportar la web (Task 10), que necesita las plantillas del `APPDATA` real y no escribe partidas.
- Juego **solo en castellano** (textos, avisos `push_warning`, líneas `Depuración: ...`). Comentarios de código y documentos en el estilo del repositorio (comentarios `##` en inglés, documentos en castellano).
- Escenas como texto: los `.tscn` solo tienen la raíz y su script; todo lo demás se construye en código. Las reglas en clases puras (`RefCounted` / `Resource`) que no tocan nodos. Sin autoloads.
- Mensajes de commit **en inglés**, sin atribución ni trailer de ningún tipo (ni `Co-Authored-By`, ni "Generated with", nada que mencione a Claude).
- **Gitflow:** la rama `feature/version-2-1-0` ya existe (creada desde `develop` en la 2.0.1). Nunca se commitea en `develop` ni en `main`; al final, merge `--no-ff` a `develop` y se borra la rama local. Sin push ni tags.

```gdscript
extends SceneTree
## Photos of world maps (the hero placed at each VIEWS point, the map camera on him) and of
## battle backdrops, with the draw calls of each one. No monster groups, no particles.
## godot --path <repo> --resolution 1280x720 --script <this> -- <out dir> [nodecor] [low]
## "nodecor": the desert without its decoration (the draw-call baseline).
## "low": the low quality flags (half the soft decoration, plain light).

const VIEWS := {
	"pradera": [Vector3(0, 0, -16), Vector3(15, 0, -15)],
	"bosque": [Vector3(-10, 0, 6)],
	"jungla": [Vector3(0, 0, 36), Vector3(-20, 0, -16)],
	"caravana": [Vector3(0, 0, 6)],
	"desierto": [Vector3(0, 0, 29), Vector3(-14, 0, 8), Vector3(24, 0, 2), Vector3(0, 0, -22),
		Vector3(-28, 0, -4), Vector3(10, 0, -6)],
}
const BACKDROPS := ["pradera", "jungla", "desierto"]
const WAIT_FRAMES := 40


func _initialize() -> void:
	_run.call_deferred()


func _run() -> void:
	var args := OS.get_cmdline_user_args()
	var out: String = args[0]
	var nodecor := args.has("nodecor")
	if args.has("low"):
		Vegetation.half_density = true
		BattleStage.plain_light = true
	ZoneParticles.enabled = false
	DirAccess.make_dir_recursive_absolute(out)
	for zone_id: String in VIEWS:
		if not zone_id in ZoneData.IDS:
			continue  # the desert before Task 2
		var points: Array = VIEWS[zone_id]
		for i in points.size():
			_forget_caches()
			var zone := ZoneData.by_id(zone_id).duplicate() as ZoneData
			zone.monster_count = 0
			if nodecor and zone_id == "desierto":
				zone.decor = []
				zone.decor_scatter = {}
			var world: Node = (load("res://scenes/world/world.tscn") as PackedScene).instantiate()
			world.zone = zone
			root.add_child(world)
			await process_frame
			world.place_player(points[i])
			for f in WAIT_FRAMES:
				await process_frame
			_save(out, "%s_%d" % [zone_id, i])
			world.queue_free()
			await process_frame
	for zone_id: String in BACKDROPS:
		if not zone_id in ZoneData.IDS:
			continue
		var stage := Node3D.new()
		root.add_child(stage)
		BattleStage.build(stage, ZoneData.by_id(zone_id))
		var camera := Camera3D.new()
		camera.fov = BattleScene.CAMERA_FOV
		stage.add_child(camera)
		camera.position = BattleScene.CAMERA_POS
		camera.look_at(BattleScene.ENEMY_POS.lerp(BattleScene.HERO_POS, 0.35) + Vector3(0, 0.8, 0))
		camera.current = true
		for f in WAIT_FRAMES:
			await process_frame
		_save(out, "combate_%s" % zone_id)
		stage.queue_free()
		await process_frame
	quit()


## The per-zone caches of Decor and WorldStage key on the zone id: a copy of a zone with other
## decoration must not reuse them.
func _forget_caches() -> void:
	Decor._placed_zone = ""
	Decor._scatter_zone = ""
	Decor._placed_grid_zone = ""
	Decor._solid_grid_zone = ""
	WorldStage._layout_zone = ""
	WorldStage._tree_zone = ""
	WorldStage._chest_zone = ""


func _save(out: String, shot: String) -> void:
	var calls := Performance.get_monitor(Performance.RENDER_TOTAL_DRAW_CALLS_IN_FRAME)
	root.get_texture().get_image().save_png(out.path_join(shot + ".png"))
	print("%s: %d llamadas de dibujo" % [shot, calls])
```

```gdscript
extends SceneTree
## Puts <before>/<x>.png and <after>/<x>.png side by side in <out>/<x>.png (left: before).
## godot --headless --path <repo> --script <this> -- <before> <after> <out>


func _initialize() -> void:
	var args := OS.get_cmdline_user_args()
	DirAccess.make_dir_recursive_absolute(args[2])
	for file in DirAccess.get_files_at(args[1]):
		if not file.ends_with(".png") or not FileAccess.file_exists(args[0].path_join(file)):
			continue
		var before := Image.load_from_file(args[0].path_join(file))
		var after := Image.load_from_file(args[1].path_join(file))
		var both := Image.create(before.get_width() + after.get_width(),
			maxi(before.get_height(), after.get_height()), false, before.get_format())
		both.blit_rect(before, Rect2i(Vector2i.ZERO, before.get_size()), Vector2i.ZERO)
		after.convert(before.get_format())
		both.blit_rect(after, Rect2i(Vector2i.ZERO, after.get_size()), Vector2i(before.get_width(), 0))
		both.save_png(args[2].path_join(file))
	quit()
```

(El disco del parche va 0,005 m por debajo de su sitio para quedar bajo el camino, que está a 0,01.) `_add_path()` y `_add_disc()` usan el color de la zona:

```gdscript
static func _add_path(root: Node3D, zone: ZoneData) -> void:
	var points := zone.path_points
	for i in points.size() - 1:
		var a := points[i]
		var b := points[i + 1]
		var d := b - a
		var mesh := PlaneMesh.new()
		mesh.size = Vector2(zone.path_half_width * 2.0, d.length())
		var mid := (a + b) / 2.0 + Vector3(0, 0.01, 0)
		var yaw := rad_to_deg(atan2(d.x, d.z))

		_add_disc(root, points[i], zone.path_half_width, zone.path_color)
	for disc in zone.safe_discs:
```

Las dunas, junto a `_add_solid_model()`:

```gdscript
## A dune: a half-buried ellipsoid of the zone's hill_color, and a static body with a row of
## upright cylinders along its long axis filling DUNE_COLLIDER_SHARE of its footprint (the
## hero walks round it; Jolt does not take scaled shapes well, so no squashed sphere).
static func _add_dune(root: Node3D, zone: ZoneData, dune: Array) -> void:
	var center: Vector3 = dune[0]
	var radii: Vector3 = dune[1]
	BattleStage.add_mesh(root, BattleStage.unit_sphere(), zone.hill_color, center, radii)
	var foot := dune_footprint(dune)
	var along_x := foot.x >= foot.y
	var long := maxf(foot.x, foot.y)
	var short := minf(foot.x, foot.y)
	var count := maxi(1, ceili(long / short))
	var body := StaticBody3D.new()
	body.position = Vector3(center.x, 0.0, center.z)
	root.add_child(body)
	for i in count:
		var offset := 0.0 if count == 1 \
			else lerpf(short - long, long - short, float(i) / (count - 1))
		var cylinder := CylinderShape3D.new()
		cylinder.radius = short * DUNE_COLLIDER_SHARE
		cylinder.height = maxf(0.5, center.y + radii.y)
		var shape := CollisionShape3D.new()
		shape.shape = cylinder
		shape.position = Vector3(offset if along_x else 0.0, cylinder.height / 2.0,
			0.0 if along_x else offset)
		body.add_child(shape)


## Half the size of a dune on the ground (x, z): its radii cut at the ground plane.
static func dune_footprint(dune: Array) -> Vector2:
	var center: Vector3 = dune[0]
	var radii: Vector3 = dune[1]
	var k := sqrt(maxf(0.0, 1.0 - pow(center.y / radii.y, 2.0)))
	return Vector2(radii.x * k, radii.z * k)


## True within margin of a dune's footprint (an ellipse on the ground).
static func near_dune(zone: ZoneData, pos: Vector3, margin: float) -> bool:
	for dune: Array in zone.dunes:
		if _in_dune(dune, pos, margin):
			return true
	return false


static func _in_dune(dune: Array, pos: Vector3, margin: float) -> bool:
	var foot := dune_footprint(dune)
	var d := Vector2(pos.x - dune[0].x, pos.z - dune[0].z)
	return pow(d.x / (foot.x + margin), 2.0) + pow(d.y / (foot.y + margin), 2.0) < 1.0


## One line for every dune of zone on something the hero must walk or reach: the path and the
## safe discs, the autoplay route, an entry, an exit, a chest or a well, a figure, the boss.
## WorldStage.build() prints them as warnings: any of them is a fault of the zone's data.
static func dune_problems(zone: ZoneData) -> PackedStringArray:
	var lines := PackedStringArray()
	for dune: Array in zone.dunes:
		var what := _dune_blocks(zone, dune)
		if what != "":
			lines.append("Duna en (%.1f, %.1f) pisa %s" % [dune[0].x, dune[0].z, what])
	return lines


static func _dune_blocks(zone: ZoneData, dune: Array) -> String:
	var points := zone.path_points
	for i in points.size() - 1:
		if _segment_in_dune(dune, points[i], points[i + 1], zone.path_half_width + DUNE_MARGIN):
			return "el camino"
	for disc in zone.safe_discs:
		if _in_dune(dune, disc[0], disc[1] + DUNE_MARGIN):
			return "el camino"
	var route := zone.autoplay_route
	for i in route.size():
		if _segment_in_dune(dune, route[i], route[(i + 1) % route.size()], DUNE_MARGIN):
			return "la ruta automática"
	for entry: Array in zone.entries.values():
		if _in_dune(dune, entry[0], DUNE_MARGIN):
			return "una entrada"
	for exit: Dictionary in zone.all_exits():
		if _in_dune(dune, exit.pos, DUNE_MARGIN):
			return "una salida"
	for id in Chests.of_zone(zone.id):
		if _in_dune(dune, Chests.CHESTS[id].pos, DUNE_MARGIN):
			return "un cofre"
	for well: Dictionary in zone.wells:
		if _in_dune(dune, well.pos, DUNE_MARGIN):
			return "un pozo"
	for npc: Dictionary in zone.npcs:
		if _in_dune(dune, npc.pos, DUNE_MARGIN):
			return "una figura"
	if zone.boss_id != "" and _in_dune(dune, zone.boss_pos, DUNE_MARGIN + 2.5):
		return "el jefe"
	return ""


static func _segment_in_dune(dune: Array, a: Vector3, b: Vector3, margin: float) -> bool:
	var steps := maxi(1, ceili(a.distance_to(b) / DUNE_SAMPLE_STEP))
	for i in steps + 1:
		if _in_dune(dune, a.lerp(b, float(i) / steps), margin):
			return true
	return false
```

```gdscript
## The desert: a low wind (noise low-passed twice at 350 Hz) swelling slowly, and the hiss
## of the sand it carries (the band between 2500 and 6000 Hz), loudest at the top of each
## swell.
static func _sand_wind(count: int, rng: RandomNumberGenerator) -> PackedFloat32Array:
	var out := PackedFloat32Array()
	out.resize(count)
	var low_a := _pole(350.0)
	var top_a := _pole(6000.0)
	var bottom_a := _pole(2500.0)
	var l1 := 0.0
	var l2 := 0.0
	var top := 0.0
	var bottom := 0.0
	for i in count:
		var t := float(i) / MusicLibrary.RATE
		var x := rng.randf_range(-1.0, 1.0)
		l1 += low_a * (x - l1)
		l2 += low_a * (l1 - l2)
		top += top_a * (x - top)
		bottom += bottom_a * (x - bottom)
		var swell := 0.5 + 0.3 * sin(TAU * 0.09 * t) + 0.2 * sin(TAU * 0.23 * t + 2.0)
		out[i] = 2.0 * l2 * swell + 0.4 * (top - bottom) * pow(swell, 3.0)
	return out


## A gust of sand: 1.5 to 2.5 s of noise through a band-pass that rises from 2500 Hz to its
## top and falls back, swelling and dying away.
static func _sand_gust(rng: RandomNumberGenerator) -> PackedFloat32Array:
	var count := int(rng.randf_range(1.5, 2.5) * MusicLibrary.RATE)
	var peak := rng.randf_range(3800.0, 5000.0)
	var out := PackedFloat32Array()
	out.resize(count)
	var band := PackedFloat32Array()
	var x1 := 0.0
	var x2 := 0.0
	var y1 := 0.0
	var y2 := 0.0
	for i in count:
		var u := float(i) / count
		if i % 32 == 0:
			band = _band(lerpf(2500.0, peak, sin(PI * u)), 3.0)
		var x := rng.randf_range(-1.0, 1.0)
		var y := band[0] * x + band[1] * x2 - band[2] * y1 - band[3] * y2
		x2 = x1
		x1 = x
		y2 = y1
		y1 = y
		out[i] = y * pow(sin(PI * u), 1.5)
	return out
```

```gdscript
func _figure_with_report(kind: String, id: String) -> int:
```
