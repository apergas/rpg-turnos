```gdscript
extends SceneTree
## Photos of world maps (the hero placed at each VIEWS point, the map camera on him) and of
## battle backdrops, with the draw calls of each one. No monster groups, no particles.
## godot --path <repo> --resolution 1280x720 --script <this> -- <out dir> [nodecor] [low]
## "nodecor": the city floors without their decoration (the draw-call baseline).
## "low": the low quality flags (half the soft decoration, plain light).

const VIEWS := {
	"pradera": [Vector3(0, 0, -16)],
	"jungla": [Vector3(0, 0, 36)],
	"desierto": [Vector3(-14, 0, 8)],
	"caravana": [Vector3(0, 0, 6)],
	"mina_1": [Vector3(0, 0, 12), Vector3(-4, 0, -4)],
	"castillo_patio": [Vector3(0, 0, 16), Vector3(12, 0, 0)],
	"montana_1": [Vector3(0, 0, 18.5), Vector3(4, 0, 4)],
	"ciudad_1": [Vector3(0, 0, 21), Vector3(0, 0, -12), Vector3(-24, 0, -16),
		Vector3(-16, 0, 12), Vector3(16, 0, 4)],
	"ciudad_2": [Vector3(0, 0, 17), Vector3(-12, 0, 8), Vector3(0, 0, -8),
		Vector3(0, 0, -11)],
}
## Zone ids whose battle backdrop is photographed ("combate_<zone id>").
const BACKDROPS := ["pradera", "desierto", "castillo_patio", "montana_1", "ciudad_1",
	"ciudad_2"]
const WAIT_FRAMES := 40


func _initialize() -> void:
	_run.call_deferred()


func _run() -> void:
	var args := OS.get_cmdline_user_args()
	var out: String = args[0]
	var nodecor := args.has("nodecor")
	if args.has("low"):
		Vegetation.half_density = true
		BattleStage.plain_light = true
	ZoneParticles.enabled = false
	DirAccess.make_dir_recursive_absolute(out)
	for zone_id: String in VIEWS:
		if not zone_id in ZoneData.IDS:
			continue  # the city before Task 2
		var points: Array = VIEWS[zone_id]
		for i in points.size():
			_forget_caches()
			var zone := ZoneData.by_id(zone_id).duplicate() as ZoneData
			zone.monster_count = 0
			if nodecor and zone_id.begins_with("ciudad"):
				zone.decor = []
			var world: Node = (load("res://scenes/world/world.tscn") as PackedScene).instantiate()
			world.zone = zone
			root.add_child(world)
			await process_frame
			world.place_player(points[i])
			for f in WAIT_FRAMES:
				await process_frame
			_save(out, "%s_%d" % [zone_id, i])
			world.queue_free()
			await process_frame
	for zone_id: String in BACKDROPS:
		if not zone_id in ZoneData.IDS:
			continue
		var zone := ZoneData.by_id(zone_id)
		var stage := Node3D.new()
		root.add_child(stage)
		BattleStage.build(stage, zone)
		var camera := Camera3D.new()
		camera.fov = BattleScene.CAMERA_FOV
		stage.add_child(camera)
		camera.position = BattleScene.CAMERA_POS
		camera.look_at(BattleScene.ENEMY_POS.lerp(BattleScene.HERO_POS, 0.35) + Vector3(0, 0.8, 0))
		camera.current = true
		for f in WAIT_FRAMES:
			await process_frame
		_save(out, "combate_%s" % zone_id)
		stage.queue_free()
		await process_frame
	quit()


## The per-zone caches of Decor and WorldStage key on the zone id: a copy of a zone with other
## decoration must not reuse them.
func _forget_caches() -> void:
	Decor._placed_zone = ""
	Decor._scatter_zone = ""
	Decor._placed_grid_zone = ""
	Decor._solid_grid_zone = ""
	WorldStage._layout_zone = ""
	WorldStage._tree_zone = ""
	WorldStage._chest_zone = ""


func _save(out: String, shot: String) -> void:
	var calls := Performance.get_monitor(Performance.RENDER_TOTAL_DRAW_CALLS_IN_FRAME)
	root.get_texture().get_image().save_png(out.path_join(shot + ".png"))
	print("%s: %d llamadas de dibujo" % [shot, calls])
```

```gdscript
extends SceneTree
## Puts <before>/<x>.png and <after>/<x>.png side by side in <out>/<x>.png (left: before).
## godot --headless --path <repo> --script <this> -- <before> <after> <out>


func _initialize() -> void:
	var args := OS.get_cmdline_user_args()
	DirAccess.make_dir_recursive_absolute(args[2])
	for file in DirAccess.get_files_at(args[1]):
		if not file.ends_with(".png") or not FileAccess.file_exists(args[0].path_join(file)):
			continue
		var before := Image.load_from_file(args[0].path_join(file))
		var after := Image.load_from_file(args[1].path_join(file))
		var both := Image.create(before.get_width() + after.get_width(),
			maxi(before.get_height(), after.get_height()), false, before.get_format())
		both.blit_rect(before, Rect2i(Vector2i.ZERO, before.get_size()), Vector2i.ZERO)
		after.convert(before.get_format())
		both.blit_rect(after, Rect2i(Vector2i.ZERO, after.get_size()), Vector2i(before.get_width(), 0))
		both.save_png(args[2].path_join(file))
	quit()
```

    cx, cy = (W - 1) // 2, (H - 1) // 2

    def walk(c):
        ch = rows[c[1]][c[0]]
        return ch not in "#P" and (ch != "D" or door_open)

    seen = {start}
    queue = deque([start])
    while queue:
        x, y = queue.popleft()
        for dx, dy in ((1, 0), (-1, 0), (0, 1), (0, -1)):
            n = (x + dx, y + dy)
            if 0 <= n[0] < W and 0 <= n[1] < H and n not in seen and walk(n):
                seen.add(n)
                queue.append(n)
    floor = [(x, y) for y in range(H) for x in range(W) if rows[y][x] != "#"]
    print(name, "reja abierta" if door_open else "reja cerrada",
          "inalcanzables:", [c + (rows[c[1]][c[0]],) for c in floor if c not in seen])

    def cell(p):
        return (math.floor(p[0] / 4 + cx + 0.5), math.floor(p[1] / 4 + cy + 0.5))

    bad = []
    for i in range(len(route or []) - 1):
        a, b = route[i], route[i + 1]
        n = max(2, int(math.dist(a, b) / 0.25))
        for k in range(n + 1):
            p = (a[0] + (b[0] - a[0]) * k / n, a[1] + (b[1] - a[1]) * k / n)
            if any(not walk(cell((p[0] + ox, p[1] + oy)))
                   for ox, oy in ((0.45, 0), (-0.45, 0), (0, 0.45), (0, -0.45))):
                bad.append((i, a, b))
                break
    print(" tramos que pisan muro:", bad)
    for label, p in marks:
        c = cell(p)
        print("  %s %s -> celda %s %r" % (label, p, c, rows[c[1]][c[0]]))

## sparks turning inside it (one mesh, one draw call) and a red light. Angry, it turns fast

	_time += delta
	_spin.rotation.z += _speed * delta
	_light.light_energy = _energy * (1.0 + 0.12 * sin(_time * 5.0))

## CALM_TIME change (a map built with the flag already set).
func set_calm(on: bool, instant := false) -> void:
	_calm = on
	if _spin == null:
		return  # _ready() applies it
	if _tween:
		_tween.kill()
	var speed := SPIN_CALM if on else SPIN
	var energy := ENERGY_CALM if on else ENERGY
	var disc := DISC_CALM if on else DISC
	var spark := SPARK_CALM if on else SPARK
	if instant or not is_inside_tree():
		_speed = speed
		_energy = energy
		_disc_material.albedo_color = disc
		_spark_material.albedo_color = spark
		return

	_tween.tween_property(self, "_speed", speed, CALM_TIME)
	_tween.tween_property(self, "_energy", energy, CALM_TIME)
	_tween.tween_property(_disc_material, "albedo_color", disc, CALM_TIME)

	tool.begin(Mesh.PRIMITIVE_TRIANGLES)
	for i in SPARKS:

		tool.append_from(sphere, 0, Transform3D(Basis(), Vector3(cos(angle) * r,
			sin(angle) * r, 0.0)))

	mat.albedo_color = color
	mat.disable_fog = true
	return mat

	var x1 := 0.0

	var y2 := 0.0
	for i in count:
		var t := float(i) / MusicLibrary.RATE
		if i % 32 == 0:
			band = _band(650.0 + 220.0 * sin(TAU * 0.06 * t) + 80.0 * sin(TAU * 0.19 * t + 2.0),
				14.0)
		var x := rng.randf_range(-1.0, 1.0)
		var y := band[0] * x + band[1] * x2 - band[2] * y1 - band[3] * y2
		x2 = x1
		x1 = x
		y2 = y1
		y1 = y
		low += low_a * (x - low)
		var gust := pow(0.5 + 0.5 * sin(TAU * 0.125 * t), 2.0)
		out[i] = 2.6 * y * (0.2 + 0.8 * gust) + 0.35 * low
	return out


## The portal quarter's deep rumble, added to the ruins' wind after the loop's seam is made:
## sines at 41, 55 and 82 Hz that throb every 2 s, as loud as the wind's peak. Each makes
## whole cycles in LOOP seconds, so the loop stays seamless.
static func _add_rumble(samples: PackedFloat32Array) -> void:
	var peak := 0.0
	for v in samples:
		peak = maxf(peak, absf(v))
	for i in samples.size():
		var t := float(i) / MusicLibrary.RATE
		var throb := 0.55 + 0.45 * sin(TAU * 0.5 * t)
		samples[i] += peak * throb * (0.5 * sin(TAU * 41.0 * t) + 0.3 * sin(TAU * 55.0 * t)
			+ 0.15 * sin(TAU * 82.0 * t))


## Stones falling in the ruins: 3 to 6 short knocks (noise through a band-pass at 900 to
## 1800 Hz, with a fast decay), each softer than the last.
static func _rubble(rng: RandomNumberGenerator) -> PackedFloat32Array:
	var out := PackedFloat32Array()
	var gain := 1.0
	for n in rng.randi_range(3, 6):
		var count := int(rng.randf_range(0.04, 0.08) * MusicLibrary.RATE)
		var band := _band(rng.randf_range(900.0, 1800.0), 6.0)
		var x1 := 0.0
		var x2 := 0.0
		var y1 := 0.0
		var y2 := 0.0
		for i in count:
			var u := float(i) / count
			var x := rng.randf_range(-1.0, 1.0)
			var y := band[0] * x + band[1] * x2 - band[2] * y1 - band[3] * y2
			x2 = x1
			x1 = x
			y2 = y1
			y1 = y
			out.append(y * gain * pow(1.0 - u, 3.0))
		_rest(out, rng.randf_range(0.05, 0.14))
		gain *= rng.randf_range(0.6, 0.85)
	return out


## A groan from the portal: 1.5 to 2.5 s of a sine sliding down to 45 Hz with its octave,
## swelling and dying away.
static func _groan(rng: RandomNumberGenerator) -> PackedFloat32Array:
	var count := int(rng.randf_range(1.5, 2.5) * MusicLibrary.RATE)
	var top := rng.randf_range(62.0, 78.0)
	var out := PackedFloat32Array()
	out.resize(count)
	var phase := 0.0
	for i in count:
		var u := float(i) / count
		phase += lerpf(top, 45.0, u) / MusicLibrary.RATE
		out[i] = (sin(TAU * phase) + 0.4 * sin(TAU * 2.0 * phase)) * pow(sin(PI * u), 1.5)
	return out
```

static func _build_city(root: Node3D, zone: ZoneData) -> void:
	var ground := PlaneMesh.new()
	ground.size = Vector2(80, 80)
	add_mesh(root, ground, CITY_FLOOR, Vector3.ZERO)
	for i in CASTLE_WALLS.size():
		var wall: Array = CASTLE_WALLS[i]

	_add_backdrop_decor(root, zone)

La figura, y una ayuda que junta muchas piezas en una sola malla (una llamada de dibujo por material, porque la figura también está en el mapa), junto a `_build_rat()`:

```gdscript
## The Devorador del portal: a dark faceted mass about 3.2 m tall with lumps on it, six
## mouths round it (dark red lips, black throats and white teeth, each kind merged into one
## mesh), tentacles at its foot, and in front a great eye (white, a red iris and a slit) under
## a heavy lid, half closed until its second phase; a red light glows in front of it. Faces -Z.
func _build_devourer(data: Dictionary) -> void:
	_devourer = Node3D.new()
	_devourer.scale = Vector3.ONE * data.scale
	_body.add_child(_devourer)
	_add_mesh(_devourer, LowPoly.blob(DEVOURER_PROFILE, 7), data.color, Vector3.ZERO, 0.9)
	var lumps: Array[Transform3D] = []
	for i in 5:
		var a := TAU * i / 5.0 + 0.4
		lumps.append(Transform3D(Basis.from_scale(Vector3.ONE * (0.55 + 0.1 * (i % 3))),
			Vector3(sin(a) * 1.1, 0.9 + 0.45 * (i % 3), -cos(a) * 1.1)))
	_add_mesh(_devourer, _merged(_sphere(1.0), lumps), data.dark_color, Vector3.ZERO, 0.9)
	var lips: Array[Transform3D] = []
	var throats: Array[Transform3D] = []
	var teeth: Array[Transform3D] = []
	for mouth: Array in DEVOURER_MOUTHS:
		var a := deg_to_rad(mouth[0])
		var out := Vector3(sin(a), 0, -cos(a))
		var frame := Transform3D(Basis.looking_at(out, Vector3.UP),
			out * 1.3 + Vector3(0, mouth[1], 0))
		lips.append(frame * Transform3D(Basis.from_scale(Vector3(0.42, 0.18, 0.15)),
			Vector3.ZERO))
		throats.append(frame * Transform3D(Basis.from_scale(Vector3(0.34, 0.1, 0.1)),
			Vector3(0, 0, -0.06)))
		for x in [-0.24, -0.08, 0.08, 0.24]:
			teeth.append(frame * Transform3D(Basis(Vector3.RIGHT, PI), Vector3(x, 0.07, -0.12)))
			teeth.append(frame * Transform3D(Basis(), Vector3(x + 0.08, -0.07, -0.12)))
	_add_mesh(_devourer, _merged(_sphere(1.0), lips), data.lip_color, Vector3.ZERO, 0.7)
	_add_mesh(_devourer, _merged(_sphere(1.0), throats), Color(0.02, 0.0, 0.02), Vector3.ZERO)
	_add_mesh(_devourer, _merged(_cone(0.04, 0.13), teeth), data.tooth_color, Vector3.ZERO, 0.5)
	var tentacles: Array[Transform3D] = []
	for i in 6:
		var a := TAU * i / 6.0 + 0.3
		var out := Vector3(sin(a), 0, -cos(a))
		tentacles.append(Transform3D(Basis.looking_at(out, Vector3.UP)
			* Basis(Vector3.RIGHT, deg_to_rad(-80.0)), out * 1.45 + Vector3(0, 0.15, 0)))
	_add_mesh(_devourer, _merged(_cone(0.2, 1.1), tentacles), data.dark_color, Vector3.ZERO)
	var eye := Vector3(0, 1.75, -1.1)
	_add_mesh(_devourer, _sphere(0.55), data.sclera_color, eye, 0.4)
	_add_mesh(_devourer, _sphere(0.28), data.iris_color, eye + Vector3(0, 0, -0.42))
	(_devourer.get_child(-1) as Node3D).scale = Vector3(1.0, 1.0, 0.4)
	_add_mesh(_devourer, _sphere(0.13), Color(0.02, 0.0, 0.02), eye + Vector3(0, 0, -0.5))
	(_devourer.get_child(-1) as Node3D).scale = Vector3(0.4, 1.0, 0.4)  # the slit
	_eyelid = Node3D.new()
	_eyelid.position = eye
	_devourer.add_child(_eyelid)
	_add_mesh(_eyelid, _hemisphere(0.62, 0.62), data.color, Vector3.ZERO, 0.9)
	_eyelid.rotation_degrees.x = EYELID_HALF
	var glow := OmniLight3D.new()
	glow.light_color = data.glow_color
	glow.light_energy = 1.2
	glow.omni_range = 5.0
	glow.position = Vector3(0, 1.5, -2.2)
	_devourer.add_child(glow)


## mesh repeated at each of xforms, merged into one mesh (one draw call).
static func _merged(mesh: Mesh, xforms: Array[Transform3D]) -> ArrayMesh:
	var tool := SurfaceTool.new()
	tool.begin(Mesh.PRIMITIVE_TRIANGLES)
	for xform in xforms:
		tool.append_from(mesh, 0, xform)
	return tool.commit()
```

Si sale al revés, cambia el signo de `EYELID_OPEN`. Las bocas miran hacia fuera con `Basis.looking_at(out)`: su −Z es `out`, así que los dientes en −Z local salen de la boca; compruébalo en la foto. Si el brillo de `glow` ilumina de más a los luchadores, bájalo a 0,8.)

## grows by DEVOURER_GROWTH; its bar goes up with it.
func _devourer_wakes() -> void:

	t.tween_property(_devourer, "scale", _devourer.scale * DEVOURER_GROWTH, 0.9) \

static func effect_text(id: String) -> String:
	var parts := PackedStringArray()
	if immunity(id) != "":
		parts.append(IMMUNE_TEXTS.get(immunity(id), "Inmune"))
	if bonus(id) > 0 or immunity(id) == "":
		parts.append("%s +%d" % [stat_name(id), bonus(id)])

## Every mesh of model in HOUSE_TINT times its own colours (its texture stays).
func _darken(model: Node3D) -> void:
	for node in model.find_children("*", "MeshInstance3D", true, false):
		var mi := node as MeshInstance3D
		for s in mi.mesh.get_surface_count():
			var mat := mi.get_active_material(s)
			if mat is BaseMaterial3D:
				var dark := mat.duplicate() as BaseMaterial3D
				dark.albedo_color *= HOUSE_TINT
				mi.set_surface_override_material(s, dark)

## After the v2's ending (the cruce scene ends under black): the game is finished in the v2
## and saved in the Caravana, which can still be walked; then the screen beyond the portal,
## the v2's credits and "Continuará", and Game goes back to the title. Game stays busy, so no
## window, ☰ or prompt shows over them.
func _finish_v2() -> void:
	MusicLibrary.build(MusicLibrary.BEYOND)  # under black
	state.story.finished_v2 = true
	_switch_zone("caravana", "")
	_world.freeze(true)
	_autosave(false)
	var numbers := _v2_numbers()
	_world.set_active(false)
	_audio.stop_ambient()
	_audio.play_music(MusicLibrary.BEYOND)
	var beyond := BeyondScreen.new()
	beyond.setup(state.loadout.worn(Equipment.WEAPON), state.loadout.worn(Equipment.SHIELD),
		state.loadout.worn(Equipment.HELMET), state.ally_id)
	add_child(beyond)
	await _fade_to(0.0)
	await beyond.run()
	await _fade_to(1.0)
	beyond.queue_free()
	var layer := CanvasLayer.new()
	layer.layer = 11
	add_child(layer)
	var credits := CreditsScreen.new()
	credits.end_text = CreditsScreen.TO_BE_CONTINUED
	layer.add_child(credits)
	await _fade_to(0.0)
	await credits.run(_v2_credits_sections(numbers))
	if _autoplay:
		print("Juego completado (v2): Nv %d, tiempo en la caravana %s, tiempo de juego %s, misiones de la caravana %d/%d"
			% [state.hero.level, numbers.time, _clock(play_time), state.quests.done_of(true),
			QuestLog.total_of(true)])
		print("Familias completas: %d/%d" % [Bestiary.complete_count(state),
			Bestiary.families(state).size()])
		print("Derrotas: %d" % _defeats)
	quit_to_title.emit()


## The v2's time and battles: since the hero left with the caravan (story.v2_start), or the
## whole game's for a save that was already in the caravan before 2.3.0 ("since" false).
func _v2_numbers() -> Dictionary:
	var start: Array = state.story.v2_start
	var since := not start.is_empty()
	return {
		"time": _clock(play_time - (float(start[0]) if since else 0.0)),
		"battles": state.battles - (int(start[1]) if since else 0),
		"since": since,
	}


## The v2's credits: the caravan's story, this journey's numbers, who made the game.
func _v2_credits_sections(numbers: Dictionary) -> Array:
	var journey: Array[String] = []
	if state.ally_id != "":
		journey.append("Compañero: %s" % Allies.display_name(state.ally_id))

		% numbers.time)

		% numbers.battles)
	journey.append("Misiones de la caravana: %d/%d" % [state.quests.done_of(true),
		QuestLog.total_of(true)])
	journey.append("Nivel: %d" % state.hero.level)
	return [
		["La caravana del portal", ["La historia de %s" % state.hero.display_name]],
		["Tu viaje", journey],
		["Creado por", [AUTHOR]],
		["Motor", ["Godot Engine 4.7"]],
		["Modelos 3D", ["KayKit, de Kay Lousberg (CC0)", "Nature Kit, de Kenney (CC0)"]],
		["Música y sonido", ["Generados por el propio juego"]],
		["Gracias por jugar", []],
	]
